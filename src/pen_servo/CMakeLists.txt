cmake_minimum_required(VERSION 3.16)
project(pen_servo LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
add_library(pen_servo INTERFACE)
target_include_directories(pen_servo INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_features(pen_servo INTERFACE cxx_std_17)
install(TARGETS pen_servo EXPORT export_pen_servo)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY include DESTINATION share/${PROJECT_NAME})
install(DIRECTORY config firmware DESTINATION share/${PROJECT_NAME}
        PATTERN build EXCLUDE PATTERN sdkconfig EXCLUDE PATTERN sdkconfig.old EXCLUDE
        PATTERN managed_components EXCLUDE)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_export_targets(export_pen_servo HAS_LIBRARY_TARGET)
ament_export_include_directories(include)
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_servo_state test/test_servo_state.cpp)
  target_link_libraries(test_servo_state pen_servo)
  add_test(NAME pen_servo_state COMMAND test_servo_state)
endif()
ament_package()
