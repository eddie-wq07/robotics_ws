cmake_minimum_required(VERSION 3.16)
project(pen_execution)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(pen_interfaces REQUIRED)
find_package(controller_manager_msgs REQUIRED)
find_package(control_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(flexiv_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(OpenSSL REQUIRED)
add_library(pen_job src/job.cpp)
target_compile_features(pen_job PUBLIC cxx_std_17)
target_include_directories(pen_job PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(pen_job PUBLIC yaml-cpp OpenSSL::Crypto)
add_executable(executor src/executor.cpp)
target_compile_options(executor PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(executor pen_job)
ament_target_dependencies(executor rclcpp rclcpp_action pen_interfaces controller_manager_msgs control_msgs
  geometry_msgs sensor_msgs diagnostic_msgs flexiv_msgs std_msgs std_srvs)
install(TARGETS pen_job executor ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_job test/test_job.cpp)
  target_link_libraries(test_job pen_job)
endif()
ament_package()
