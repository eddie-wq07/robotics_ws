cmake_minimum_required(VERSION 3.16)
project(pen_description)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
ament_python_install_package(${PROJECT_NAME})
set(PEN_ASSET_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../assets/meshes" CACHE PATH "Canonical pen mesh directory")
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/check_meshes.py ${PEN_ASSET_ROOT}
                RESULT_VARIABLE mesh_result OUTPUT_VARIABLE mesh_output ERROR_VARIABLE mesh_error)
if(NOT mesh_result EQUAL 0)
  message(FATAL_ERROR "Canonical pen assets invalid: ${mesh_error}${mesh_output}")
endif()
install(DIRECTORY urdf config DESTINATION share/${PROJECT_NAME})
install(FILES assets/tylenol/tylenol.glb assets/tylenol/manifest.json
        DESTINATION share/${PROJECT_NAME}/assets/tylenol)
install(DIRECTORY "${PEN_ASSET_ROOT}/" DESTINATION share/${PROJECT_NAME}/meshes FILES_MATCHING PATTERN "*.stl" PATTERN "manifest.json")
install(PROGRAMS scripts/render_description.py DESTINATION lib/${PROJECT_NAME} RENAME render_description)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_calibration test/test_calibration.py APPEND_ENV PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR})
endif()
ament_package()
